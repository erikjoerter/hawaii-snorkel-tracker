    <!-- Logic Script -->

    <script>

        // Initialize Lucide Icons

        lucide.createIcons();


        const spotSelect = document.getElementById('spot-select');

        

        // Fetch data when user changes the dropdown or loads the page

        spotSelect.addEventListener('change', fetchMarineData);

        window.addEventListener('DOMContentLoaded', fetchMarineData);


        async function fetchMarineData() {

            const selectedOption = spotSelect.options[spotSelect.selectedIndex];

            const lat = selectedOption.getAttribute('data-lat');

            const lng = selectedOption.getAttribute('data-lng');

            const spotName = selectedOption.text;


            // Reset UI State to loading

            document.getElementById('spot-title').innerText = spotName;

            document.getElementById('status-text').innerText = "Analyzing...";

            document.getElementById('status-banner').className = "bg-slate-400 text-white p-6 text-center";


            try {

                // Fetch real-time wave heights and wind data from Open-Meteo Marine API

                const url = `https://open-meteo.com{lat}&longitude=${lng}&current=wave_height,wind_wave_height&hourly=wave_height&temperature_unit=fahrenheit&wind_speed_unit=mph&length_unit=inch`;

                

                // Fetch basic weather (wind & temp) alongside marine data

                const weatherUrl = `https://open-meteo.com{lat}&longitude=${lng}&current=temperature_2m,wind_speed_10m&temperature_unit=fahrenheit&wind_speed_unit=mph`;


                const [marineRes, weatherRes] = await Promise.all([fetch(url), fetch(weatherUrl)]);

                const marineData = await marineRes.json();

                const weatherData = await weatherRes.json();


                // Extract metrics

                const waveHeightInches = marineData.current.wave_height || 0;

                const waveHeightFeet = (waveHeightInches / 12).toFixed(1);

                const windSpeed = weatherData.current.wind_speed_10m.toFixed(1);

                const waterTemp = weatherData.current.temperature_2m.toFixed(1); // Approximated via coastal air temp for simplicity


                // Update UI Metrics

                document.getElementById('metric-waves').innerText = `${waveHeightFeet} ft`;

                document.getElementById('metric-wind').innerText = `${windSpeed} mph`;

                document.getElementById('metric-temp').innerText = `${waterTemp} °F`;


                // Run Safety Evaluation Algorithm

                evaluateConditions(waveHeightFeet, windSpeed);


            } catch (error) {

                console.error("Error fetching data:", error);

                document.getElementById('status-text').innerText = "Data Error";

                document.getElementById('advice-text').innerText = "Failed to load real-time ocean data. Please check back later.";

            }

        }


        function evaluateConditions(waves, wind) {

            const banner = document.getElementById('status-banner');

            const statusText = document.getElementById('status-text');

            const adviceText = document.getElementById('advice-text');


            // CRITICAL THRESHOLDS FOR SNORKELING

            // Waves > 4ft = Dangerous for snorkeling (churns sand, creates undertow)

            // Waves > 2.5ft = Fair, visibility reduced

            // Wind > 15mph = Choppy, surface currents strong


            if (waves >= 4.0 || wind >= 20) {

                // DANGEROUS CONDITIONS

                banner.className = "bg-red-500 text-white p-6 text-center transition-colors duration-300";

                statusText.innerText = "🔴 POOR / DANGEROUS";

                adviceText.innerText = `High surf alert! Waves are around ${waves}ft with winds at ${wind}mph. This creates strong undertows and visibility will be near zero due to churned up sand. It is highly recommended to stay out of the water today.`;

            } else if (waves >= 2.5 || wind >= 13) {

                // CAUTION CONDITIONS

                banner.className = "bg-amber-500 text-white p-6 text-center transition-colors duration-300";

                statusText.innerText = "🟡 FAIR / CAUTION";


Use code with caution.

adviceText.innerText = Moderate conditions. Wave heights are ${waves}ft. Snorkeling is possible, but expect a surface chop and reduced underwater visibility. Beginners should stick very close to the shore or a lifeguard tower.;

} else {

// IDEAL CONDITIONS

banner.className = "bg-emerald-500 text-white p-6 text-center transition-colors duration-300";

statusText.innerText = "🟢 EXCELLENT CONDITIONS";

adviceText.innerText = Perfect day for snorkeling! Calm seas (${waves}ft) and low winds mean crystal-clear underwater visibility. Enjoy the coral reefs and marine life safely!;

}
